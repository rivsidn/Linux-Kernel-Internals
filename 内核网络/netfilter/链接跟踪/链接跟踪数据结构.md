## 结构体

### nf_conn

对应一条会话.


### nf_conntrack_tuple_hash

nf_conn{} 中的五元组.

可以通过hash 嵌入到hash 表、unconfirmed、dying 表中.


### nf_conntrack_tuple

对应五元组，在五元组基础上多加了一个 dir .

```c
struct nf_conntrack_man {
	union nf_inet_addr u3;              //src IP
	union nf_conntrack_man_proto u;     //src Port
	/* Layer 3 protocol */
	u_int16_t l3num;                    //三层协议号(IPv4、IPv6)
};

struct nf_conntrack_tuple {
	struct nf_conntrack_man src;

	/* These are the parts of the tuple which are fixed. */
	struct {
		union nf_inet_addr u3;          //dst IP
		union {                         //dst Port
			/* Add other protocols here. */
			__be16 all;

			struct {
				__be16 port;
			} tcp;
			struct {
				__be16 port;
			} udp;
			struct {
				u_int8_t type, code;
			} icmp;
			struct {
				__be16 port;
			} dccp;
			struct {
				__be16 port;
			} sctp;
			struct {
				__be16 key;
			} gre;
		} u;

		/* The protocol. */
		u_int8_t protonum;              //四层协议号(TCP、UDP)

		/* The direction (for tuplehash) */
		u_int8_t dir;                   //方向(original、reply)
	} dst;
};

```
