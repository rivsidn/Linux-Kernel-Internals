
# Linux-2.6.12

## 应用示例-定时器删除

```c
int del_timer(struct timer_list *timer)
{
	unsigned long flags;
	tvec_base_t *base;

	check_timer(timer);

repeat:
 	base = timer->base;
	if (!base)
		return 0;
	spin_lock_irqsave(&base->lock, flags);
	if (base != timer->base) {
		spin_unlock_irqrestore(&base->lock, flags);
		goto repeat;
	}
	list_del(&timer->entry);
	/* Need to make sure that anybody who sees a NULL base also sees the list ops */
	smp_wmb();
	timer->base = NULL;
	spin_unlock_irqrestore(&base->lock, flags);

	return 1;
}

```

```
● 这里的 merge（合并）
  会带来一些重要影响，特别是在并发编程中：

  什么是内存访问合并？

  CPU 可以将多个小的内存访问合并成一个大的访问：

  // 原始代码
  X = *A;        // 读取 4 字节
  Y = *(A + 4);  // 读取下一个 4 字节

  // CPU 可能合并成
  {X, Y} = LOAD_8BYTES(A);  // 一次读取 8 字节

  潜在的问题和影响

  1. 原子性假设被破坏

  // 错误的假设：两次独立的 4 字节读取
  struct {
      uint32_t counter;    // A
      uint32_t flag;       // A+4
  } data;

  // Thread 1：
  data.counter = 100;
  data.flag = 1;  // 标记数据就绪

  // Thread 2：
  if (data.flag == 1) {       // 可能被合并！
      use(data.counter);       // 可能读到旧值
  }

  问题：CPU 可能一次性读取 8 字节，在 flag
  更新前就读取了两个值。

  2. 设备寄存器访问问题

  // MMIO 寄存器（硬件设备）
  volatile uint32_t *status_reg =
  (void*)0x10000000;
  volatile uint32_t *data_reg   =
  (void*)0x10000004;

  // 期望：先读状态，再读数据
  uint32_t status = *status_reg;  // 
  触发硬件状态更新
  uint32_t data = *data_reg;      // 
  读取更新后的数据

  // 如果被合并成一次 8 字节读取：
  // 硬件可能没机会在两次读取间更新数据！

  3. 锁分离失效

  struct {
      spinlock_t lock1;
      int data1;
      spinlock_t lock2;  // 相邻的锁
      int data2;
  } s;

  // CPU 可能合并读取 lock1 和 lock2
  // 导致假共享和性能问题

  4. 内存屏障失效

  // 期望的顺序
  X = *A;
  memory_barrier();
  Y = *(A + 4);

  // 但如果 CPU 合并成一次读取：
  {X, Y} = LOAD {*A, *(A + 4)};
  // 屏障失去意义！

  写入合并的影响

  *A = X;
  *(A + 4) = Y;

  // 可能被合并成
  STORE_8BYTES(A) = {X, Y};

  问题场景：

  // Thread 1：
  *A = 1;        // 设置标志
  *(A + 4) = 2;  // 设置数据

  // Thread 2：
  if (*A == 1) {  // 检查标志
      use(*(A + 4));  // 使用数据
  }

  // 如果写入被合并，两个值同时可见
  // 破坏了"先设置标志，后设置数据"的语义

  如何避免合并？

  1. 使用 READ_ONCE/WRITE_ONCE

  X = READ_ONCE(*A);
  Y = READ_ONCE(*(A + 4));
  // 防止编译器合并，但 CPU 仍可能合并

  2. 使用 volatile

  volatile uint32_t *p = A;
  X = *p;
  Y = *(p + 1);
  // 告诉编译器不要优化

  3. 使用内存屏障

  X = *A;
  smp_mb();  // 完整的内存屏障
  Y = *(A + 4);
  // 强制分开执行

  4. 使用原子操作

  X = atomic_read(A);
  Y = atomic_read(A + 4);
  // 保证独立的原子访问

  实际影响

  这种合并主要影响：
  1. 设备驱动（MMIO 访问顺序很重要）
  2. 无锁数据结构（依赖特定的访问顺序）
  3. 多线程同步（假设的原子性被破坏）
  4. 调试（行为与预期不同）

  这就是为什么内核编程需要特别小心，不能假设内存访
  问会按照代码顺序独立执行。

```
